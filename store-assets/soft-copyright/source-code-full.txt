export class GatewayHttpError extends Error {
  public statusCode: number;
  constructor(statusCode: number) {
    super(`Gateway request failed: ${statusCode}`);
    this.statusCode = statusCode;
  }
}
export class GatewayNetworkError extends Error {
  public detail: string;
  constructor(detail: string = '') {
    super('Gateway network request failed');
    this.detail = detail;
  }
}
export function isGatewayAuthenticationError(error: Object): boolean {
  if (!(error instanceof GatewayHttpError)) {
    return false;
  }
  return error.statusCode === 401 || error.statusCode === 403;
}
export function isGatewayNetworkError(error: Object): boolean {
  return error instanceof GatewayNetworkError;
}

class SystemGatewayTransport implements GatewayTransport {
  private session: rcp.Session;
  private caContent: string;
  constructor(caContent: string = '') {
    this.caContent = caContent;
    const timeoutCfg: rcp.Timeout = {
      connectMs: 30000,
      transferMs: 30000
    };
    if (caContent.length > 0) {
      console.info(`[SystemGatewayTransport] Creating session with CA cert, length=${caContent.length}`);
      const caCert: rcp.CertificateAuthority = { content: this.caContent };
      const sessionCfg: rcp.SessionConfiguration = {
        requestConfiguration: {
          transfer: {
            timeout: timeoutCfg
          },
          security: { remoteValidation: caCert }
        }
      };
      this.session = rcp.createSession(sessionCfg);
    } else {
      console.info('[SystemGatewayTransport] Creating session WITHOUT CA cert (default trust)');
      const sessionCfg: rcp.SessionConfiguration = {
        requestConfiguration: {
          transfer: {
            timeout: timeoutCfg
          }
        }
      };
      this.session = rcp.createSession(sessionCfg);
    }
  }
  async request(url: string, options: RequestOptions): Promise<GatewayHttpResponse> {
    const req = new rcp.Request(url, options.method);
    const headers: Record<string, string> = {};
    if (options.headers.contentType.length > 0) {
      headers['content-type'] = options.headers.contentType;
    }
    if (options.headers.authorization.length > 0) {
      headers['authorization'] = options.headers.authorization;
    }
    req.headers = headers;
    if (options.body !== undefined && options.body.length > 0) {
      req.content = options.body;
    }
    try {
      const response = await this.session.fetch(req);
      const result: string = JSON.stringify(response.toJSON());
      return {
        statusCode: response.statusCode,
        result: result
      };
    } catch (error) {
      const detail: string = typeof error === 'object' && error !== null ? JSON.stringify(error) : String(error);
      console.error(`[GatewayTransport] request failed: url=${url}, method=${options.method}, error=${detail}`);
      throw new GatewayNetworkError(detail);
    }
  }
}
export class HttpGatewayClient implements GatewayClient {
  private transport: GatewayTransport;
  constructor(transport: GatewayTransport = new SystemGatewayTransport()) {
    this.transport = transport;
  }
  static createWithCert(certContent: string): HttpGatewayClient {
    return new HttpGatewayClient(new SystemGatewayTransport(certContent));
  }
  async register(request: RegisterRequest): Promise<RegisterResponse> {
    const response = await this.transport.request(joinUrl(request.baseUrl, '/api/auth/register'), {
      method: 'POST',
      headers: {
        contentType: 'application/json',
        authorization: ''
      },
      body: JSON.stringify({
        deviceId: request.deviceId,
        publicKey: request.publicKey,
        registrationCode: request.registrationCode,
        deviceName: request.deviceName
      })
    });
    ensureSuccess(response);
    const body = JSON.parse(response.result) as RegisterResponseBody;
    return {
      activationToken: body.activationToken,
      challenge: body.challenge
    };
  }
  async activate(request: ActivateRequest): Promise<LoginResponse> {
    const response = await this.transport.request(joinUrl(request.baseUrl, '/api/auth/activate'), {
      method: 'POST',
      headers: {
        contentType: 'application/json',
        authorization: ''
      },
      body: JSON.stringify({
        deviceId: request.deviceId,
        activationToken: request.activationToken,
        signedChallenge: request.signedChallenge
      })
    });
    ensureSuccess(response);
    const body = JSON.parse(response.result) as LoginResponseBody;
    return {
      accessToken: body.accessToken,
      refreshToken: body.refreshToken,
      sessionName: 'opencode-main',
      expiresIn: 900
    };
  }
  async challenge(request: ChallengeRequest): Promise<ChallengeResponse> {
    const response = await this.transport.request(joinUrl(request.baseUrl, '/api/auth/challenge'), {
      method: 'POST',
      headers: {
        contentType: 'application/json',
        authorization: ''
      },
      body: JSON.stringify({
        deviceId: request.deviceId
      })
    });
    ensureSuccess(response);
    const body = JSON.parse(response.result) as ChallengeResponseBody;
    return {
      nonce: body.nonce,
      expiresAt: body.expiresAt
    };
  }
  async verify(request: VerifyRequest): Promise<LoginResponse> {
    const response = await this.transport.request(joinUrl(request.baseUrl, '/api/auth/verify'), {
      method: 'POST',
      headers: {
        contentType: 'application/json',
        authorization: ''
      },
      body: JSON.stringify({
        deviceId: request.deviceId,
        nonce: request.nonce,
        signature: request.signature
      })
    });
    ensureSuccess(response);
    const body = JSON.parse(response.result) as LoginResponseBody;
    return {
      accessToken: body.accessToken,
      refreshToken: body.refreshToken,
      sessionName: 'opencode-main',
      expiresIn: 900
    };
  }
  async refresh(request: RefreshRequest): Promise<RefreshResponse> {
    const response = await this.transport.request(joinUrl(request.baseUrl, '/api/auth/refresh'), {
      method: 'POST',
      headers: {
        contentType: 'application/json',
        authorization: ''
      },
      body: JSON.stringify({
        refreshToken: request.refreshToken
      })
    });
    ensureSuccess(response);
    const body = JSON.parse(response.result) as RefreshResponseBody;
    return {
      accessToken: body.accessToken
    };
  }
  async logout(request: LogoutRequest): Promise<void> {
    try {
      await this.transport.request(joinUrl(request.baseUrl, '/api/auth/logout'), {
        method: 'POST',
        headers: {
          contentType: 'application/json',
          authorization: ''
        },
        body: JSON.stringify({
          refreshToken: request.refreshToken
        })
      });
    } catch {
    }
  }
  async checkHealth(baseUrl: string): Promise<boolean> {
    const response = await this.transport.request(joinUrl(baseUrl, '/health'), {
      method: 'GET',
      headers: {
        contentType: '',
        authorization: ''
      }
    });
    if (response.statusCode !== 200) {
      return false;
    }
    try {
      const body = JSON.parse(response.result) as HealthResponseBody;
      return body.status === 'ok';
    } catch {
      return false;
    }
  }
  async listResumes(request: ListResumesRequest): Promise<ResumeItem[]> {
    const url = joinUrl(request.baseUrl, OPENCODE_RESUMES_PATH);
    console.info(`[listResumes] 请求 URL: ${url}`);
    const response = await this.transport.request(url, {
      method: 'GET',
      headers: {
        contentType: '',
        authorization: `Bearer ${request.accessToken}`
      }
    });
    ensureSuccess(response);
    const body = JSON.parse(response.result) as ListResumesResponseBody;
    return body.items;
  }
  async startResume(request: StartResumeRequest): Promise<ResumeSession> {
    const response = await this.transport.request(joinUrl(request.baseUrl, OPENCODE_RESUMES_PATH), {
      method: 'POST',
      headers: {
        contentType: 'application/json',
        authorization: `Bearer ${request.accessToken}`
      },
      body: JSON.stringify({
        cwd: normalizeString(request.cwd)
      })
    });
    ensureSuccess(response);
    return toResumeSession(JSON.parse(response.result) as ResumeThreadResponseBody, '');
  }
  async resumeThread(request: ResumeThreadRequest): Promise<ResumeSession> {
    const response = await this.transport.request(
      joinUrl(request.baseUrl, `${OPENCODE_RESUMES_PATH}/${encodeURIComponent(request.threadId)}/resume`),
      {
        method: 'POST',
        headers: {
          contentType: 'application/json',
          authorization: `Bearer ${request.accessToken}`
        },
        body: JSON.stringify({
          threadId: request.threadId
        })
      }
    );
    ensureSuccess(response);
    return toResumeSession(JSON.parse(response.result) as ResumeThreadResponseBody, request.threadId);
  }
  async sendResumeMessage(request: SendResumeMessageRequest): Promise<ResumeSession> {
    const response = await this.transport.request(
      joinUrl(request.baseUrl, `${OPENCODE_RESUMES_PATH}/${encodeURIComponent(request.threadId)}/messages`),
      {
        method: 'POST',
        headers: {
          contentType: 'application/json',
          authorization: `Bearer ${request.accessToken}`
        },
        body: JSON.stringify({
          message: request.message
        })
      }
    );
    ensureSuccess(response);
    return toResumeSession(JSON.parse(response.result) as ResumeThreadResponseBody, request.threadId);
  }
  async renameResume(request: RenameResumeRequest): Promise<void> {
    const response = await this.transport.request(
      joinUrl(request.baseUrl, `${OPENCODE_RESUMES_PATH}/${encodeURIComponent(request.threadId)}/name`),
      {
        method: 'POST',
        headers: {
          contentType: 'application/json',
          authorization: `Bearer ${request.accessToken}`
        },
        body: JSON.stringify({
          name: request.name
        })
      }
    );
    ensureSuccess(response);
  }
  async deleteResume(request: ResumeMutationRequest): Promise<void> {
    const response = await this.transport.request(
      joinUrl(request.baseUrl, `${OPENCODE_RESUMES_PATH}/${encodeURIComponent(request.threadId)}`),
      {
        method: 'DELETE',
        headers: {
          contentType: '',
          authorization: `Bearer ${request.accessToken}`
        }
      }
    );
    ensureSuccess(response);
  }
}

function toResumeSession(body: ResumeThreadResponseBody, fallbackThreadId: string): ResumeSession {
  const thread = body.thread;
  const status = body.status !== undefined ? body.status : thread?.status;
  const title = normalizeString(body.title) || normalizeString(thread?.name) || normalizeString(thread?.preview) ||
    fallbackThreadId;
  return {
    threadId: normalizeString(body.threadId) || normalizeString(thread?.id) || fallbackThreadId,
    title,
    cwd: normalizeString(body.cwd) || normalizeString(thread?.cwd),
    status: normalizeStatus(status),
    turns: normalizeTurns(body.turns !== undefined ? body.turns : thread?.turns)
  };
}
function normalizeTurns(value: ResumeTurn[] | undefined): ResumeTurn[] {
  if (!Array.isArray(value)) {
    return [];
  }
  return value.map((turn: ResumeTurn, turnIndex: number) => {
    const turnId = normalizeString(turn.id) || `turn-${turnIndex + 1}`;
    const normalized: ResumeTurn = {
      id: turnId,
      status: normalizeString(turn.status) || 'unknown',
      startedAt: normalizeString(turn.startedAt),
      completedAt: normalizeString(turn.completedAt),
      items: normalizeItems(turn.items, turnId)
    };
    return normalized;
  }).filter((turn: ResumeTurn) => turn.items.length > 0 || isActiveResumeTurnStatus(turn.status));
}
function normalizeItems(value: ResumeContentItem[], turnId: string): ResumeContentItem[] {
  if (!Array.isArray(value)) {
    return [];
  }
  return value.map((item: ResumeContentItem, itemIndex: number) => {
    const normalized: ResumeContentItem = {
      id: normalizeString(item.id) || `${turnId}:item-${itemIndex + 1}`,
      role: normalizeString(item.role) || 'system',
      kind: normalizeString(item.kind) || 'message',
      text: normalizeString(item.text),
      status: normalizeString(item.status)
    };
    return normalized;
  }).filter((item: ResumeContentItem) => item.text.length > 0);
}
function normalizeString(value: string | undefined): string {
  if (typeof value !== 'string') {
    return '';
  }
  return value.trim();
}
function normalizeStatus(value: string | ThreadStatus | undefined): string {
  if (typeof value === 'string') {
    return value;
  }
  if (value && typeof value.type === 'string') {
    return value.type;
  }
  return 'unknown';
}
function isActiveResumeTurnStatus(status: string): boolean {
  return status === 'inProgress' || status === 'running' || status === 'queued' || status === 'in_progress';
}
function joinUrl(baseUrl: string, path: string): string {
  return `${baseUrl.replace(/\/+$/, '')}${path}`;
}
function ensureSuccess(response: GatewayHttpResponse): void {
  if (response.statusCode < 200 || response.statusCode >= 300) {
    throw new GatewayHttpError(response.statusCode);
  }
}

    threadId: '',
    title: '',
    cwd: '',
    status: '',
    turns: []
  };
}
function appendResumeMdInlineSegment(segments: ResumeMdInlineSegment[], text: string, kind: string, href: string): void {
  if (text.length === 0) {
    return;
  }
  const lastIndex = segments.length - 1;
  if (lastIndex >= 0 && segments[lastIndex].kind === kind && segments[lastIndex].href === href) {
    segments[lastIndex] = {
      id: segments[lastIndex].id,
      text: `${segments[lastIndex].text}${text}`,
      kind,
      href
    };
    return;
  }
  segments.push({
    id: `${segments.length}-${kind}`,
    text,
    kind,
    href
  });
}
function findNextResumeMdBoldMarker(text: string, start: number): ResumeMdInlineMarker {
  for (let index = start; index < text.length - 1; index++) {
    if (text.charAt(index) === '*' && text.charAt(index + 1) === '*') {
      const contentStart = index + 2;
      const closeIndex = text.indexOf('**', contentStart);
      if (closeIndex > contentStart) {
        return {
          start: index,
          end: closeIndex + 2,
          kind: RESUME_MD_BLOCK_BOLD,
          content: text.substring(contentStart, closeIndex),
          href: ''
        };
      }
    }
  }
  return { start: -1, end: -1, kind: RESUME_MD_BLOCK_TEXT, content: '', href: '' };
}
function findNextResumeMdItalicMarker(text: string, start: number): ResumeMdInlineMarker {
  for (let index = start; index < text.length; index++) {
    if (text.charAt(index) !== '*') {
      continue;
    }
    if (index + 1 < text.length && text.charAt(index + 1) === '*') {
      continue;
    }
    if (index - 1 >= start && text.charAt(index - 1) === '*') {
      continue;
    }
    const closeIndex = text.indexOf('*', index + 1);
    if (closeIndex > index + 1 && text.charAt(closeIndex + 1) !== '*') {
      return {
        start: index,
        end: closeIndex + 1,
        kind: RESUME_MD_BLOCK_ITALIC,
        content: text.substring(index + 1, closeIndex),
        href: ''
      };
    }
  }
  return { start: -1, end: -1, kind: RESUME_MD_BLOCK_TEXT, content: '', href: '' };
}
function findNextResumeMdCodeMarker(text: string, start: number): ResumeMdInlineMarker {
  for (let index = start; index < text.length; index++) {
    if (text.charAt(index) !== '`') {
      continue;
    }
    const closeIndex = text.indexOf('`', index + 1);
    if (closeIndex > index + 1) {
      return {
        start: index,
        end: closeIndex + 1,
        kind: RESUME_MD_BLOCK_CODE,
        content: text.substring(index + 1, closeIndex),
        href: ''
      };
    }
  }
  return { start: -1, end: -1, kind: RESUME_MD_BLOCK_TEXT, content: '', href: '' };
}
function findNextResumeMdLinkMarker(text: string, start: number): ResumeMdInlineMarker {
  for (let index = start; index < text.length; index++) {
    if (text.charAt(index) !== '[') {
      continue;
    }
    const linkStart = text.indexOf('](', index + 1);
    if (linkStart < 0) {
      continue;
    }
    const hrefEnd = text.indexOf(')', linkStart + 2);
    if (hrefEnd < 0) {
      continue;
    }
    const label = text.substring(index + 1, linkStart);
    const href = text.substring(linkStart + 2, hrefEnd);
    if (label.length > 0 && href.length > 0) {
      return {
        start: index,
        end: hrefEnd + 1,
        kind: RESUME_MD_BLOCK_LINK,
        content: label,
        href
      };
    }
  }
  return { start: -1, end: -1, kind: RESUME_MD_BLOCK_TEXT, content: '', href: '' };
}
function findNextResumeMdInlineMarker(text: string, start: number): ResumeMdInlineMarker {
  const bold = findNextResumeMdBoldMarker(text, start);
  const italic = findNextResumeMdItalicMarker(text, start);
  const code = findNextResumeMdCodeMarker(text, start);
  const link = findNextResumeMdLinkMarker(text, start);
  let result = bold;
  if (result.start < 0 || (italic.start >= 0 && italic.start < result.start)) {
    result = italic;
  }
  if (result.start < 0 || (code.start >= 0 && code.start < result.start)) {
    result = code;
  }
  if (result.start < 0 || (link.start >= 0 && link.start < result.start)) {
    result = link;
  }
  return result;
}
function resolveResumeMarkdownInline(text: string): ResumeMdInlineSegment[] {
  const segments: ResumeMdInlineSegment[] = [];
  let cursor = 0;
  while (cursor < text.length) {
    const match = findNextResumeMdInlineMarker(text, cursor);
    if (match.start < 0) {
      appendResumeMdInlineSegment(segments, text.substring(cursor), RESUME_MD_BLOCK_TEXT, '');
      return segments;
    }
    if (match.start > cursor) {
      appendResumeMdInlineSegment(segments, text.substring(cursor, match.start), RESUME_MD_BLOCK_TEXT, '');
    }
    appendResumeMdInlineSegment(segments, match.content, match.kind, match.href);
    cursor = match.end;
  }
  return segments;
}
function resolveResumeMdHeadingLevel(line: string): number {
  if (line.startsWith('### ')) {
    return 3;
  }
  if (line.startsWith('## ')) {
    return 2;
  }
  if (line.startsWith('# ')) {
    return 1;
  }
  return 0;
}
function resolveResumeMdListText(line: string): string {
  const trimmed = line.trim();
  if (trimmed.startsWith('- ') || trimmed.startsWith('* ')) {
    return trimmed.substring(2);
  }
  const orderedMatch = trimmed.match(new RegExp('^\\d+[.)]\\s+'));
  if (orderedMatch !== null) {
    return trimmed.substring(orderedMatch[0].length);
  }
  return trimmed;
}
function resolveResumeMarkdownBlocks(text: string): ResumeMdBlock[] {
  const blocks: ResumeMdBlock[] = [];
  const lines = text.split('\n');
  let paragraphLines: string[] = [];
  let blockIndex = 0;
  const flushParagraph = (): void => {
    if (paragraphLines.length > 0) {
      const paragraphText = paragraphLines.join('\n');
      blocks.push({
        id: `md-${blockIndex++}-paragraph`,
        kind: RESUME_MD_BLOCK_PARAGRAPH,
        level: 0,
        language: '',
        text: paragraphText,
        inline: resolveResumeMarkdownInline(paragraphText)
      });
      paragraphLines = [];
    }
  };
  let index = 0;
  while (index < lines.length) {
    const line = lines[index];
    const trimmed = line.trim();
    if (trimmed.startsWith('```')) {
      flushParagraph();
      const language = trimmed.substring(3).trim();
      const codeLines: string[] = [];
      index++;
      while (index < lines.length && !lines[index].trim().startsWith('```')) {
        codeLines.push(lines[index]);
        index++;
      }
      index++;
      blocks.push({
        id: `md-${blockIndex++}-code`,
        kind: RESUME_MD_BLOCK_CODE,
        level: 0,
        language,
        text: codeLines.join('\n'),
        inline: []
      });
      continue;
    }
    const headingLevel = resolveResumeMdHeadingLevel(trimmed);
    if (headingLevel > 0) {
      flushParagraph();
      const headingText = trimmed.substring(headingLevel + 1).trim();
      blocks.push({
        id: `md-${blockIndex++}-heading`,
        kind: RESUME_MD_BLOCK_HEADING,
        level: headingLevel,
        language: '',
        text: headingText,
        inline: resolveResumeMarkdownInline(headingText)
      });
      index++;
      continue;
    }
    if (trimmed.startsWith('- ') || trimmed.startsWith('* ') || new RegExp('^\\d+[.)]\\s+').test(trimmed)) {
      flushParagraph();
      const listText = resolveResumeMdListText(trimmed);
      blocks.push({
        id: `md-${blockIndex++}-list`,
        kind: RESUME_MD_BLOCK_LIST,
        level: 0,
        language: '',
        text: listText,
        inline: resolveResumeMarkdownInline(listText)
      });
      index++;
      continue;
    }
    if (trimmed.startsWith('>')) {
      flushParagraph();
      const quoteText = trimmed.substring(1).trim();
      blocks.push({
        id: `md-${blockIndex++}-quote`,
        kind: RESUME_MD_BLOCK_QUOTE,
        level: 0,
        language: '',
        text: quoteText,
        inline: resolveResumeMarkdownInline(quoteText)
      });
      index++;
      continue;
    }
    if (trimmed === '---' || trimmed === '***') {
      flushParagraph();
      blocks.push({
        id: `md-${blockIndex++}-divider`,
        kind: RESUME_MD_BLOCK_DIVIDER,
        level: 0,
        language: '',
        text: '',
        inline: []
      });
      index++;
      continue;
    }
    if (trimmed.length === 0) {
      flushParagraph();
      index++;
      continue;
    }
    paragraphLines.push(line);
    index++;
  }
  flushParagraph();
  return blocks;
}

        this.resumeDarkMode = false;

      this.applyResumeDarkMode(false);

    }

  private async loadCardResumes(connection: SavedConnection): Promise<void> {
    const refreshId = ++this.connectionResumeRefreshId;
    const cached = this.resumesCache.get(connection.id);
    this.resumesConnectionId = connection.id;
    this.resumes = cached !== undefined ? cached : [];
    this.resumesFeedback = '';
    this.resumeActionFeedback = '';
    this.openingResumeId = '';
    this.resumeActionMenuId = '';
    this.resumesLoading = cached === undefined;
    this.autoConnectingConnectionId = connection.id;
    this.clearCardAutoConnectFailure(connection.id);
    this.updateConnectionStatus(connection, 'offline', false);
    let attempt = 0;
    let activeConnection: SavedConnection | undefined = connection;
    let lastError: Error | undefined;
    while (attempt < CARD_AUTO_CONNECT_MAX_ATTEMPTS) {
      attempt++;
      const currentConnection: SavedConnection | undefined = activeConnection !== undefined ? activeConnection :
        this.connectionById(connection.id);
      if (currentConnection === undefined || !shouldApplyConnectionResumeRefresh(connection.id, this.resumesConnectionId,
        refreshId, this.connectionResumeRefreshId)) {
        break;
      }
      const refreshedConnection: SavedConnection | undefined =
        await this.ensureConnectionCredentialsForCard(currentConnection);
      if (!shouldApplyConnectionResumeRefresh(connection.id, this.resumesConnectionId, refreshId,
        this.connectionResumeRefreshId)) {
        break;
      }
      if (refreshedConnection !== undefined) {
        activeConnection = refreshedConnection;
      }
      const requestConnection: SavedConnection = activeConnection !== undefined ? activeConnection : currentConnection;
      try {
        const loadedResumes = await this.gatewayClient.listResumes({
          baseUrl: `${requestConnection.scheme}://${requestConnection.host}:${requestConnection.port}`,
          accessToken: requestConnection.accessTokenRef
        });
        if (!shouldApplyConnectionResumeRefresh(connection.id, this.resumesConnectionId, refreshId,
          this.connectionResumeRefreshId)) {
          break;
        }
        this.setResumes(connection.id, loadedResumes);
        this.resumesFeedback = this.resumes.length === 0 ? '暂无会话' : '';
        this.clearCardAutoConnectFailure(connection.id);
        this.updateConnectionStatus(requestConnection, 'online', false);
        this.resumesLoading = false;
        this.autoConnectingConnectionId = '';
        return;
      } catch (error) {
        const err = error as Error;
        console.error(`[loadCardResumes] 第${attempt}次尝试失败: name=${err.name}, message=${err.message}, stack=${err.stack}`);
        lastError = err;
        if (!shouldApplyConnectionResumeRefresh(connection.id, this.resumesConnectionId, refreshId,
          this.connectionResumeRefreshId)) {
          break;
        }
        if (isGatewayNetworkError(err)) {
          break;
        }
        if (isGatewayAuthenticationError(err)) {
          const retryConnection: SavedConnection | undefined =
            await this.refreshConnectionCredentials(requestConnection, 'online', false);
          if (!shouldApplyConnectionResumeRefresh(connection.id, this.resumesConnectionId, refreshId,
            this.connectionResumeRefreshId)) {
            break;
          }
          if (retryConnection !== undefined) {
            activeConnection = retryConnection;
          }
        } else {
          break;
        }
      }
    }
    if (shouldApplyConnectionResumeRefresh(connection.id, this.resumesConnectionId, refreshId,
      this.connectionResumeRefreshId)) {
      this.resumes = [];
      if (lastError !== undefined) {
        if (isGatewayNetworkError(lastError)) {
          this.resumesFeedback = `网络连接失败: ${lastError.message}`;
        } else if (isGatewayAuthenticationError(lastError)) {
          const authError = lastError as GatewayHttpError;
          this.resumesFeedback = `认证失败(${authError.statusCode})`;
        } else {
          this.resumesFeedback = `连接失败: ${lastError.message}`;
        }
      } else {
        this.resumesFeedback = '无法自动连接服务器';
      }
      this.autoConnectFailedConnectionId = connection.id;
      this.updateConnectionStatus(connection, 'offline', false);
      this.resumesLoading = false;
      this.autoConnectingConnectionId = '';
    }
  }
  private prefetchConnectionSwiperReleaseTarget(index: number, position: number): void {
    const targetIndex = resolveConnectionSwiperReleaseIndex(index, this.connections.length, position);
    if (targetIndex === index) {
      return;
    }
    const selection: ConnectionCarouselSelection = resolveConnectionSelectionBySwiperIndex(this.connections, targetIndex);
    if (selection.id.length === 0 || selection.id === this.connectionSwiperPrefetchedTargetId) {
      return;
    }
    const connection = this.connectionById(selection.id);

      return;

    this.prefetchCardResumes(connection);
  }
  private shouldApplyConnectionResumePrefetch(connectionId: string, prefetchId: number): boolean {
    return this.connectionResumePrefetchIds.get(connectionId) === prefetchId &&
      this.connectionById(connectionId) !== undefined;
  }
  private async prefetchCardResumes(connection: SavedConnection): Promise<void> {
    if (connection.id.length === 0 || this.connectionResumePrefetchIds.has(connection.id)) {
      return;
    }
    const prefetchId = ++this.connectionResumePrefetchId;
    this.connectionResumePrefetchIds.set(connection.id, prefetchId);
    let attempt = 0;
    let activeConnection: SavedConnection | undefined = connection;
    try {
      while (attempt < CARD_AUTO_CONNECT_MAX_ATTEMPTS) {
        attempt++;
        const currentConnection: SavedConnection | undefined = activeConnection !== undefined ? activeConnection :
          this.connectionById(connection.id);
        if (currentConnection === undefined ||
          !this.shouldApplyConnectionResumePrefetch(connection.id, prefetchId)) {
          break;
        }
        const refreshedConnection: SavedConnection | undefined =
          await this.ensureConnectionCredentialsForCard(currentConnection);
        if (!this.shouldApplyConnectionResumePrefetch(connection.id, prefetchId)) {
          break;
        }
        if (refreshedConnection !== undefined) {
          activeConnection = refreshedConnection;
        }
        const requestConnection: SavedConnection = activeConnection !== undefined ? activeConnection : currentConnection;
        try {
          const loadedResumes = await this.gatewayClient.listResumes({
            baseUrl: `${requestConnection.scheme}://${requestConnection.host}:${requestConnection.port}`,
            accessToken: requestConnection.accessTokenRef
          });
          if (!this.shouldApplyConnectionResumePrefetch(connection.id, prefetchId)) {
            break;
          }
          this.resumesCache.set(connection.id, loadedResumes);
          if (this.resumesConnectionId === connection.id) {
            this.resumes = loadedResumes;
            this.resumesFeedback = loadedResumes.length === 0 ? '暂无会话' : '';
            this.resumesLoading = false;
          }
          this.clearCardAutoConnectFailure(connection.id);
          this.updateConnectionStatus(requestConnection, 'online', false);
          return;
        } catch (error) {
          if (!this.shouldApplyConnectionResumePrefetch(connection.id, prefetchId)) {
            break;
          }
          if (isGatewayAuthenticationError(error as Error)) {
            const retryConnection: SavedConnection | undefined =
              await this.refreshConnectionCredentials(requestConnection, 'online', false);
            if (!this.shouldApplyConnectionResumePrefetch(connection.id, prefetchId)) {
              break;
            }
            if (retryConnection !== undefined) {
              activeConnection = retryConnection;
            }
          }
        }
      }
      if (this.shouldApplyConnectionResumePrefetch(connection.id, prefetchId)) {
        this.updateConnectionStatus(connection, 'offline', false);
      }
    } finally {
      if (this.connectionResumePrefetchIds.get(connection.id) === prefetchId) {
        this.connectionResumePrefetchIds.delete(connection.id);
      }
    }
  }

  private async saveEditedConnection(): Promise<void> {
    const selected = this.selectedConnection();
    if (selected.id.length === 0) {
      this.closeDialog();
      return;
    }
    const errors = resolveConnectionFormErrors(this.draftAddress, this.draftPassword, false);
    if (errors.hasError) {
      this.applyFieldFeedback(errors);
      this.dialogFeedback = '';
      this.feedback = '';
      return;
    }
    const normalized = normalizeServerAddress(this.draftAddress);
    const nextConnectionId = generateSavedConnectionId(normalized.host);
    const identityChanged = generateSavedConnectionId(selected.host) !== nextConnectionId;
    if (hasConnectionWithSameEndpoint(this.connections, normalized.scheme, normalized.host, normalized.port,
      selected.id)) {
      this.addressFeedback = '该服务器地址已存在';
      this.dialogFeedback = '同一个 Agent 和 IP 只能添加一个连接';
      this.feedback = '';
      return;
    }
    if (identityChanged && this.draftPassword.trim().length === 0) {
      this.passwordFeedback = '更改服务器地址时需要重新输入注册码';
      this.dialogFeedback = '';
      this.feedback = '';
      return;
    }
    const name = resolveConnectionName(this.draftName, '', '', normalized.host);
    let nextAccessTokenRef: string | undefined = undefined;
    let nextRefreshTokenRef: string | undefined = undefined;
    if (this.draftPassword.trim().length > 0) {
      try {
        const keyPair = await ensureDeviceKeyPair();
        const registerResult = await this.gatewayClient.register({
          baseUrl: normalized.baseUrl,
          deviceId: this.deviceId,
          publicKey: keyPair.publicKeyPem,
          registrationCode: this.draftPassword,
          deviceName: 'HarmonyOS Phone'
        });
        const sig = await signChallenge(registerResult.challenge);
        const activateResult = await this.gatewayClient.activate({
          baseUrl: normalized.baseUrl,
          deviceId: this.deviceId,
          activationToken: registerResult.activationToken,
          signedChallenge: sig
        });
        nextAccessTokenRef = activateResult.accessToken;
        nextRefreshTokenRef = activateResult.refreshToken;
      } catch (error) {
        this.dialogFeedback = this.resolveLoginFailureText(error as Error);
        this.feedback = '';
        return;
      }
    }
    this.connections = updateSavedConnection(this.connections, {
      id: selected.id,
      name,
      agent: 'OpenCode',
      host: normalized.host,
      port: normalized.port,
      scheme: normalized.scheme,
      username: '',
      lastUsedAt: new Date().toISOString(),
      accessTokenRef: nextAccessTokenRef,
      refreshTokenRef: nextRefreshTokenRef,
      privateKeyRef: this.draftPassword.trim().length > 0 ? this.draftPassword : undefined
    });
    const updatedConnection = this.connectionById(nextConnectionId);
    if (updatedConnection !== undefined && shouldClearConnectionResumeCache(selected, updatedConnection)) {
      this.resumesCache.delete(selected.id);
    }
    if (identityChanged) {
      this.resumesCache.delete(selected.id);
      this.resumesCache.delete(nextConnectionId);
    }
    this.activeConnectionId = nextConnectionId;
    this.activeConnectionIndex = resolveConnectionIndexById(this.connections, this.activeConnectionId,
      this.activeConnectionIndex);
    this.persistConnections();
    this.closeDialog();
    this.resumesConnectionId = '';
    if (this.connections.length === 0) {
      this.resumes = [];
      this.resumesFeedback = '';
      this.feedback = '';
      return;
    }
    this.loadActiveConnectionResumes();
  }

  private async resumeSelectedThread(resume: ResumeItem): Promise<void> {

      const session: ResumeSession = await this.gatewayClient.resumeThread({
        baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
        accessToken: connection.accessTokenRef,
        threadId: resume.id
      });
      this.activeResumeSession = session;
      this.resumeStopRequested = false;
      this.resetResumeContentZoomScale();
      this.hasActiveResumeSession = true;
      this.resumeComposerVisible = true;
      this.clearResumePageTapTimer();

      this.syncResumeAutoRefresh();
      if (session.turns.length === 0) {
        this.resumeActionFeedback = '已载入会话，但没有可展示的历史内容';
      } else {
        this.resumeActionFeedback = '';
      }
      this.dialog = ConnectionDialog.NONE;
      this.updateBackAvailability();
      this.scrollActiveResumeToBottom();
    } catch (error) {
      if (isGatewayAuthenticationError(error as Error)) {
        const refreshedConnection = await this.refreshConnectionCredentials(connection);
        if (refreshedConnection !== undefined) {
          this.openingResumeId = '';
          this.resumeSelectedThread(resume);
          return;
        }
        this.promptRelogin(connection, '登录已失效，正在重新认证');
        return;
      }
      this.resumeActionFeedback = '无法打开会话';
    } finally {
      this.openingResumeId = '';
    }
  }

    this.showResumeCenterNotice('正在新建会话');
    this.scrollActiveResumeToBottom();
    try {
      const session: ResumeSession = await this.gatewayClient.startResume({
        baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
        accessToken: connection.accessTokenRef,
        cwd: DEFAULT_RESUME_CWD
      });
      if (startToken !== this.resumeRefreshToken) {
        return;
      }
      this.activeResumeSession = session;
      this.hasActiveResumeSession = true;
      this.updateConnectionStatus(connection, 'online');
      this.resumeComposerVisible = true;
      this.clearResumePageTapTimer();
      this.syncResumeWaitingIndicator();
      this.syncResumeAutoRefresh();
      this.resumeActionFeedback = '';
      this.showResumeCenterNotice('已新建会话，可直接发送消息');
      this.updateBackAvailability();
      this.scrollActiveResumeToBottom();
      this.loadResumesWithoutClearingSession(connection);
    } catch (error) {
      if (startToken !== this.resumeRefreshToken) {
        return;
      }
      if (isGatewayAuthenticationError(error as Error)) {
        const refreshedConnection = await this.refreshConnectionCredentials(connection);
        if (refreshedConnection !== undefined) {
          this.resumeStarting = false;
          this.clearActiveResumeSession();
          this.startNewResumeForConnection(refreshedConnection);
          return;
        }
        this.clearActiveResumeSession();
        this.promptRelogin(connection, '登录已失效，正在重新认证');
        return;
      }
      this.resumeActionFeedback = '新建会话失败';
    } finally {
      if (startToken === this.resumeRefreshToken) {
        this.resumeStarting = false;
      }
    }
  }
  private async loadResumesWithoutClearingSession(connection: SavedConnection): Promise<void> {

      this.resumesFeedback = '';
      this.updateConnectionStatus(connection, 'online');
    } catch (error) {
      if (!shouldApplyConnectionResumeRefresh(connection.id, this.resumesConnectionId, refreshId,
        this.connectionResumeRefreshId)) {
        return;
      }
      if (isGatewayAuthenticationError(error as Error)) {
        const refreshedConnection = await this.refreshConnectionCredentials(connection);
        if (refreshedConnection !== undefined) {
          this.loadResumesWithoutClearingSession(refreshedConnection);
          return;
        }
        this.promptRelogin(connection, '登录已失效，正在重新认证');

  private async testConnection(): Promise<void> {

    const normalized = normalizeServerAddress(this.draftAddress);
    this.testingConnection = true;
    try {
      const ok = await this.gatewayClient.checkHealth(normalized.baseUrl);
      if (ok) {
        this.connectionTestResult = '连接成功，服务器运行正常';
        this.connectionTestPassed = true;
      } else {
        this.connectionTestResult = '服务器已连接，但服务状态异常';
        this.connectionTestPassed = false;
      }
    } catch (error) {
      this.connectionTestPassed = false;
      const networkError = error as GatewayNetworkError;
      this.connectionTestResult = this.resolveTestFailureText(networkError);
    } finally {
      this.testingConnection = false;
    }

  private resolveTestFailureText(error: GatewayNetworkError): string {
    const detail: string = error.detail.length > 0 ? error.detail.toLowerCase() : '';
    if (detail.includes('certificate') || detail.includes('cert')) {
      return '证书校验失败：该服务器的证书不受信任，请联系管理员';
    }
    if (detail.includes('timeout') || detail.includes('timed out')) {
      return '连接超时：请检查服务器地址和网络';
    }
    if (detail.includes('refused') || detail.includes('econnrefused')) {
      return '连接被拒绝：端口未开放或服务未启动';
    }
    if (detail.includes('dns') || detail.includes('eai_again') || detail.includes('ehostunreach')) {
      return '无法解析主机：请检查服务器地址是否正确';
    }
    return '网络异常，无法连接服务器，请检查地址和网络';
  }
  private async addConnection(): Promise<void> {
    if (this.connectionSubmitting) {
      return;
    }
    if (!canAddConnection(this.connections)) {
      this.dialogFeedback = '最多支持 5 个连接，请删除一个连接后再添加。';
      this.feedback = '';
      return;
    }
    const errors = resolveConnectionFormErrors(this.draftAddress, this.draftPassword, true);
    if (errors.hasError) {
      this.applyFieldFeedback(errors);
      this.dialogFeedback = '';
      this.feedback = '';
      return;
    }
    const normalized = normalizeServerAddress(this.draftAddress);
    if (hasConnectionWithSameEndpoint(this.connections, normalized.scheme, normalized.host, normalized.port, '')) {
      this.addressFeedback = '该服务器地址已存在';
      this.dialogFeedback = '同一个 Agent 和 IP 只能添加一个连接';
      this.feedback = '';
      return;
    }
    const requestId = ++this.addConnectionRequestId;
    this.connectionSubmitting = true;
    let loginResponse: LoginResponse;
    console.info(`[addConnection] certContentLoaded=${this.certContentLoaded}`);
    console.info(`[addConnection] 登录 baseUrl: ${normalized.baseUrl}, scheme: ${normalized.scheme}`);
    try {
      const keyPair = await ensureDeviceKeyPair();
      const registerResult = await this.gatewayClient.register({
        baseUrl: normalized.baseUrl,
        deviceId: this.deviceId,
        publicKey: keyPair.publicKeyPem,
        registrationCode: this.draftPassword,
        deviceName: 'HarmonyOS Phone'
      });
      const sig = await signChallenge(registerResult.challenge);
      loginResponse = await this.gatewayClient.activate({
        baseUrl: normalized.baseUrl,
        deviceId: this.deviceId,
        activationToken: registerResult.activationToken,
        signedChallenge: sig
      });
    } catch (error) {
      if (requestId !== this.addConnectionRequestId) {
        return;
      }
      this.connectionSubmitting = false;
      this.dialogFeedback = this.resolveLoginFailureText(error as Error);
      this.feedback = '';
      return;
    }
    if (requestId !== this.addConnectionRequestId || this.dialog !== ConnectionDialog.ADD) {
      return;
    }
    const now = new Date().toISOString();
    const id = generateSavedConnectionId(normalized.host);
    const name = resolveConnectionName(this.draftName, loginResponse.sessionName, '', normalized.host);
    const nextConnection: SavedConnection = {
      id,
      name,
      agent: 'OpenCode',
      host: normalized.host,
      port: normalized.port,
      scheme: normalized.scheme,
      username: '',
      lastUsedAt: now,
      status: 'online',
      accessTokenRef: loginResponse.accessToken,
      refreshTokenRef: loginResponse.refreshToken,
      privateKeyRef: ''
    };
    const nextConnections = appendSavedConnection(this.connections, nextConnection);
    if (nextConnections.length === this.connections.length) {
      this.connectionSubmitting = false;
      if (!canAddConnection(this.connections)) {
        this.dialogFeedback = '最多支持 5 个连接，请删除一个连接后再添加。';
      } else {
        this.addressFeedback = '该服务器地址已存在';
        this.dialogFeedback = '同一个 Agent 和 IP 只能添加一个连接';
      }
      this.feedback = '';
      return;
    }
    this.connections = nextConnections;
    this.activeConnectionId = nextConnection.id;
    this.activeConnectionIndex = resolveConnectionIndexById(this.connections, this.activeConnectionId, 0);
    this.persistConnections();
    this.connectionSubmitting = false;
    this.closeDialog();
    this.loadCardResumes(nextConnection);
  }

  private applyConnectionCredentials(
    connection: SavedConnection,
    accessToken: string,
    refreshToken: string,
    privateKey: string,
    status: ConnectionStatus = 'online',
    touchLastUsedAt: boolean = true
  ): SavedConnection | undefined {
    this.connections = updateSavedConnection(this.connections, {
      id: connection.id,
      name: connection.name,
      agent: connection.agent,
      host: connection.host,
      port: connection.port,
      scheme: connection.scheme,
      username: connection.username,
      lastUsedAt: touchLastUsedAt ? new Date().toISOString() : connection.lastUsedAt,
      accessTokenRef: accessToken,
      refreshTokenRef: refreshToken,
      privateKeyRef: ''
    }).map((item: SavedConnection) => {
      if (item.id !== connection.id) {
        return item;
      }
      return {
        id: item.id,
        name: item.name,
        agent: item.agent,
        host: item.host,
        port: item.port,
        scheme: item.scheme,
        username: item.username,
        lastUsedAt: item.lastUsedAt,
        status,
        accessTokenRef: item.accessTokenRef,
        refreshTokenRef: item.refreshTokenRef,
        privateKeyRef: item.privateKeyRef
      };
    });
    this.keepActiveConnectionInBounds();
    this.persistConnections();
    return this.connectionById(connection.id);
  }
  private async refreshConnectionCredentials(
    connection: SavedConnection,
    status: ConnectionStatus = 'online',
    touchLastUsedAt: boolean = true
  ): Promise<SavedConnection | undefined> {
    const refreshToken = connection.refreshTokenRef.trim();
    if (refreshToken.length > 0) {
      try {
        const refreshResult = await this.gatewayClient.refresh({
          baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
          refreshToken
        });
        return this.applyConnectionCredentials(connection, refreshResult.accessToken, refreshToken, '',
          status, touchLastUsedAt);
      } catch {
      }
    }
    try {
      const nonceResult = await this.gatewayClient.challenge({
        baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
        deviceId: this.deviceId
      });
      const sig = await signChallenge(nonceResult.nonce);
      const loginResponse = await this.gatewayClient.verify({
        baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
        deviceId: this.deviceId,
        nonce: nonceResult.nonce,
        signature: sig
      });
      return this.applyConnectionCredentials(connection, loginResponse.accessToken, loginResponse.refreshToken, '',
        status, touchLastUsedAt);
    } catch (error) {
      return undefined;
    }
  }
  private async handleConnectionSwitch(connection: SavedConnection): Promise<void> {
    const intent = resolveConnectionSwitchIntent(connection.status);
    if (intent === 'turn_off') {
      this.updateConnectionStatus(connection, 'offline');
      this.feedback = `${connection.name} 已断开`;
      return;
    }
    const readinessError = resolveConnectionReadinessError(connection);
    if (readinessError.length > 0) {
      this.swipedConnectionId = '';
      this.dialogFeedback = '';
      this.feedback = readinessError;
      return;
    }
    if (intent === 'require_login') {
      const refreshedConnection = await this.refreshConnectionCredentials(connection);
      if (refreshedConnection !== undefined) {
        this.feedback = '';
        this.showResumeList(refreshedConnection);
        return;
      }
      this.selectedConnectionId = connection.id;
      this.reloginConnectionId = connection.id;
      this.dialogFeedback = '';
      this.dialog = ConnectionDialog.RELOGIN;
      this.updateBackAvailability();
      return;
    }
    this.updateConnectionStatus(connection, 'online');
    this.feedback = '';
    this.showResumeList(connection);
  }

      return;
    }
    const connection = foundConnection;
    try {
      const nonceResult = await this.gatewayClient.challenge({
        baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
        deviceId: this.deviceId
      });
      const sig = await signChallenge(nonceResult.nonce);
      const loginResponse = await this.gatewayClient.verify({
        baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
        deviceId: this.deviceId,
        nonce: nonceResult.nonce,
        signature: sig
      });
      this.applyConnectionCredentials(connection, loginResponse.accessToken, loginResponse.refreshToken, '');
      this.feedback = '';
      const refreshedConnection = this.connectionById(connection.id);
      this.dialog = ConnectionDialog.NONE;
      this.selectedConnectionId = '';
      this.reloginConnectionId = '';
      this.dialogFeedback = '';
      this.draftPassword = '';
      this.updateBackAvailability();
      if (refreshedConnection !== undefined) {
        this.loadCardResumes(refreshedConnection);
      }
    } catch (error) {
      this.dialogFeedback = this.resolveLoginFailureText(error as Error);
      this.feedback = '';
    }
  }
  private resolveLoginFailureText(error: Error): string {
    if (isGatewayNetworkError(error)) {
      const networkError = error as GatewayNetworkError;
      const detail: string = networkError.detail.length > 0 ? networkError.detail.toLowerCase() : '';
      if (detail.includes('certificate') || detail.includes('cert')) {
        return '证书校验失败：服务器证书不受信任，请联系管理员更新证书';
      }
      if (detail.includes('timeout') || detail.includes('timed out')) {
        return '连接超时：请检查服务器是否在线及网络连接';
      }
      if (detail.includes('refused') || detail.includes('econnrefused')) {
        return '连接被拒绝：请检查服务器端口是否开放';
      }
      if (detail.includes('dns') || detail.includes('eai_again') || detail.includes('ehostunreach')) {
        return '无法解析主机：请检查服务器地址是否正确';
      }
      return '网络异常，无法连接服务器，请检查地址和网络';
    }
    if (isGatewayAuthenticationError(error)) {
      const authError = error as GatewayHttpError;
      if (authError.statusCode === 401) {
        return '认证失败：注册码错误或已被使用';
      }
      return `认证失败(${authError.statusCode})：设备无权访问`;
    }
    return '服务器异常，请稍后重试';
  }

  private async sendResumeMessage(): Promise<void> {
    if (!this.hasActiveResumeSession || this.resumeMessageSending) {
      return;
    }
    if (this.resumeStarting || this.activeResumeSession.threadId.length === 0) {
      this.resumeActionFeedback = '会话创建中';
      return;
    }
    const message = this.normalizeResumeInputText(this.resumeMessageDraft).trim();
    if (message.length === 0) {
      this.resumeActionFeedback = '请输入消息';
      return;
    }
    const connection = this.selectedConnection();
    if (connection.id.length === 0) {
      this.resumeActionFeedback = '连接不可用';
      return;
    }
    this.resumeMessageSending = true;
    this.pendingResumeMessage = message;
    this.pendingResumeMessageTimestamp = resolveResumeTimestampText(new Date().toISOString());
    this.resumeMessageDraft = '';
    this.resumeJustSent = true;
    this.resumeStopRequested = false;
    this.resumeRefreshToken++;
    this.resumeActionFeedback = '';
    this.startResumeWaitingIndicator();
    this.scrollActiveResumeToBottom();
    try {
      const session: ResumeSession = await this.gatewayClient.sendResumeMessage({
        baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
        accessToken: connection.accessTokenRef,
        threadId: this.activeResumeSession.threadId,
        message
      });
      this.activeResumeSession = session;
      this.hasActiveResumeSession = true;
      this.updateConnectionStatus(connection, 'online');
      this.pendingResumeMessage = '';
      this.pendingResumeMessageTimestamp = '';
      if (session.turns.length === 0) {
        this.resumeActionFeedback = '已发送，但没有可展示的历史内容';
        this.resumeMessageSending = false;
        this.syncResumeWaitingIndicator();
        this.syncResumeAutoRefresh();
        return;
      }
      this.resumeActionFeedback = '';
      this.resumeMessageSending = false;
      this.resumeComposerVisible = true;
      this.syncResumeWaitingIndicator();
      this.syncResumeAutoRefresh();
      this.scrollActiveResumeToBottom(false);
    } catch (error) {
      if (isGatewayAuthenticationError(error as Error)) {
        const refreshedConnection = await this.refreshConnectionCredentials(connection);
        if (refreshedConnection !== undefined) {
          this.resumeMessageSending = false;
          this.resumeMessageDraft = this.pendingResumeMessage;
          this.pendingResumeMessage = '';
          this.pendingResumeMessageTimestamp = '';
          this.resumeComposerVisible = true;
          this.syncResumeWaitingIndicator();
          this.syncResumeAutoRefresh();
          this.sendResumeMessage();
          return;
        }
        this.promptRelogin(connection, '登录已失效，正在重新认证');
        this.resumeMessageSending = false;
        this.pendingResumeMessage = '';
        this.pendingResumeMessageTimestamp = '';
        this.resumeComposerVisible = true;
        this.syncResumeWaitingIndicator();
        this.syncResumeAutoRefresh();
        return;
      }
      this.resumeActionFeedback = '消息已发送，等待回复中...';
      this.resumeMessageSending = false;
      this.pendingResumeMessage = '';
      this.pendingResumeMessageTimestamp = '';
      this.resumeComposerVisible = true;
      this.syncResumeWaitingIndicator();
      this.scheduleTimeoutRetryRefresh();
    }
  }

    const delay = hasInProgress ? 3000 : 10000;

  private scheduleTimeoutRetryRefresh(): void {
    if (!this.hasActiveResumeSession) {
      return;
    }
    const threadId = this.activeResumeSession.threadId;
    const refreshToken = this.resumeRefreshToken;
    let retries = 0;
    const maxRetries = 20;
    const interval = 3000;
    const poll = () => {
      if (retries >= maxRetries || refreshToken !== this.resumeRefreshToken) {
        return;
      }
      retries++;
      this.resumeAutoRefreshTimer = setTimeout(() => {
        this.resumeAutoRefreshTimer = -1;
        this.refreshActiveResumeSilently(threadId, refreshToken);
        if (this.hasActiveResumeSession && this.resumeHasInProgressTurn(this.activeResumeSession)) {
          poll();
        }
      }, interval);
    };
    poll();
  }

  private async refreshActiveResume(): Promise<void> {
    if (!this.hasActiveResumeSession) {
      return;
    }
    const connection = this.selectedConnection();
    if (connection.id.length === 0) {
      this.resumeActionFeedback = '连接不可用';
      return;
    }
    try {
      const session: ResumeSession = await this.gatewayClient.resumeThread({
        baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
        accessToken: connection.accessTokenRef,
        threadId: this.activeResumeSession.threadId
      });
      this.activeResumeSession = session;
      this.hasActiveResumeSession = true;
      this.updateConnectionStatus(connection, 'online');

        const refreshedConnection = await this.refreshConnectionCredentials(connection);

  private async refreshActiveResumeSilently(threadId: string, refreshToken: number): Promise<void> {
    if (!this.hasActiveResumeSession || this.activeResumeSession.threadId !== threadId ||
      this.resumeRefreshToken !== refreshToken) {
      return;
    }
    const connection = this.selectedConnection();
    if (connection.id.length === 0) {
      this.syncResumeAutoRefresh();
      return;
    }
    try {
      const session: ResumeSession = await this.gatewayClient.resumeThread({
        baseUrl: `${connection.scheme}://${connection.host}:${connection.port}`,
        accessToken: connection.accessTokenRef,
        threadId
      });
      if (!this.hasActiveResumeSession || this.activeResumeSession.threadId !== threadId ||
        this.resumeRefreshToken !== refreshToken) {
        return;
      }
      this.activeResumeSession = session;
      this.hasActiveResumeSession = true;
      this.updateConnectionStatus(connection, 'online');
      if (!this.resumeHasInProgressTurn(session)) {
        this.resumeMessageSending = false;
        this.pendingResumeMessage = '';
        this.pendingResumeMessageTimestamp = '';
      }
      this.syncResumeWaitingIndicator();
      this.syncResumeAutoRefresh();
      this.scrollActiveResumeToBottom(false);
    } catch (error) {
      if (isGatewayAuthenticationError(error as Error)) {
        const refreshedConnection = await this.refreshConnectionCredentials(connection);
        if (refreshedConnection !== undefined) {
          this.refreshActiveResumeSilently(threadId, refreshToken);
          return;
        }
        this.promptRelogin(connection, '登录已失效，正在重新认证');
        return;
      }
      this.stopResumeAutoRefresh();
    }

    if (segment.style === RESUME_TEXT_STYLE_SUCCESS) {
      return TERMINAL_GREEN_COLOR;
    }
    if (segment.style === RESUME_TEXT_STYLE_WARNING) {
      return TERMINAL_AMBER_COLOR;
    }
    if (segment.style === RESUME_TEXT_STYLE_DANGER) {
      return TERMINAL_DANGER_TEXT_COLOR;
    }
    return this.resumeBodyTextColor(item);
  }
  private resumeSegmentFontWeight(segment: ResumeTextSegment): FontWeight {
    if (segment.style === RESUME_TEXT_STYLE_CODE || segment.style === RESUME_TEXT_STYLE_PATH ||
      segment.style === RESUME_TEXT_STYLE_DANGER) {
      return FontWeight.Medium;
    }
    return FontWeight.Regular;
  }
  private resumeInlineCodeStyle(text: string): string {
    if (this.resumeLooksLikePath(text)) {
      return RESUME_TEXT_STYLE_PATH;
    }
    return RESUME_TEXT_STYLE_CODE;
  }
  private resumeLooksLikePath(text: string): boolean {
    return text.indexOf('/') >= 0 || text.indexOf('\\') >= 0 ||
      text.endsWith('.ets') || text.endsWith('.ts') || text.endsWith('.js') ||
      text.endsWith('.json') || text.endsWith('.json5') || text.endsWith('.md') ||
      text.endsWith('.txt') || text.endsWith('.ps1') || text.endsWith('.bat');
  }
  private findResumeKeywordMatch(text: string, start: number, tokens: string[], style: string): ResumeKeywordMatch {
    let bestStart = -1;
    let bestEnd = -1;
    const normalized = text.toLowerCase();
    for (let index = 0; index < tokens.length; index++) {
      const matchStart = normalized.indexOf(tokens[index], start);
      if (matchStart < 0) {
        continue;
      }
      const matchEnd = matchStart + tokens[index].length;
      if (bestStart < 0 || matchStart < bestStart || (matchStart === bestStart && matchEnd > bestEnd)) {
        bestStart = matchStart;
        bestEnd = matchEnd;
      }
    }
    return {
      start: bestStart,
      end: bestEnd,
      style
    };
  }
  private nextResumeKeywordMatch(text: string, start: number): ResumeKeywordMatch {
    const danger = this.findResumeKeywordMatch(text, start, RESUME_DANGER_TOKENS, RESUME_TEXT_STYLE_DANGER);
    const warning = this.findResumeKeywordMatch(text, start, RESUME_WARNING_TOKENS, RESUME_TEXT_STYLE_WARNING);
    const success = this.findResumeKeywordMatch(text, start, RESUME_SUCCESS_TOKENS, RESUME_TEXT_STYLE_SUCCESS);
    let result = danger;
    if (result.start < 0 || (warning.start >= 0 && warning.start < result.start)) {
      result = warning;
    }
    if (result.start < 0 || (success.start >= 0 && success.start < result.start)) {
      result = success;
    }
    return result;
  }
  private appendResumeTextSegment(segments: ResumeTextSegment[], text: string, style: string): void {
    if (text.length === 0) {
      return;
    }
    const lastIndex = segments.length - 1;
    if (lastIndex >= 0 && segments[lastIndex].style === style) {
      segments[lastIndex] = {
        id: segments[lastIndex].id,
        text: `${segments[lastIndex].text}${text}`,
        style
      };
      return;
    }
    segments.push({
      id: `${segments.length}-${style}`,
      text,
      style
    });
  }
  private appendResumePlainTextSegments(segments: ResumeTextSegment[], text: string, fallbackStyle: string): void {
    let lineStart = 0;
    let inFileDiffContext = false;
    for (let index = 0; index <= text.length; index++) {
      if (index < text.length && text.charAt(index) !== '\n') {
        continue;
      }
      const lineEnd = index < text.length ? index + 1 : index;
      const line = text.substring(lineStart, lineEnd);
      inFileDiffContext = resolveResumeDiffLineContext(line, inFileDiffContext);
      const lineStyle = resolveResumeDiffLineTextStyle(line, fallbackStyle, RESUME_TEXT_STYLE_DANGER,
        RESUME_TEXT_STYLE_SUCCESS, inFileDiffContext);
      if (lineStyle !== fallbackStyle) {
        this.appendResumeTextSegment(segments, line, lineStyle);
      } else {
        this.appendResumePlainLineSegments(segments, line, fallbackStyle);
      }
      lineStart = index + 1;
    }
  }
  private appendResumePlainLineSegments(segments: ResumeTextSegment[], text: string, fallbackStyle: string): void {
    let cursor = 0;
    while (cursor < text.length) {
      const match = this.nextResumeKeywordMatch(text, cursor);
      if (match.start < 0) {
        this.appendResumeTextSegment(segments, text.substring(cursor), fallbackStyle);
        return;
      }
      if (match.start > cursor) {
        this.appendResumeTextSegment(segments, text.substring(cursor, match.start), fallbackStyle);
      }
      this.appendResumeTextSegment(segments, text.substring(match.start, match.end), match.style);
      cursor = match.end;
    }
  }
  private resumeTextSegments(item: ResumeContentItem): ResumeTextSegment[] {
    const segments: ResumeTextSegment[] = [];
    const fallbackStyle = this.resumeBodyTextStyle(item);
    let plainText = '';
    let inlineText = '';
    let inInlineCode = false;
    for (let index = 0; index < item.text.length; index++) {
      const char = item.text.charAt(index);
      if (char === '`') {
        if (inInlineCode) {
          this.appendResumeTextSegment(segments, inlineText, this.resumeInlineCodeStyle(inlineText));
          inlineText = '';
          inInlineCode = false;
        } else {
          this.appendResumePlainTextSegments(segments, plainText, fallbackStyle);
          plainText = '';
          inInlineCode = true;
        }
        continue;
      }
      if (inInlineCode) {
        inlineText = `${inlineText}${char}`;
      } else {
        plainText = `${plainText}${char}`;
      }
    }
    if (inInlineCode) {
      this.appendResumePlainTextSegments(segments, `\`${inlineText}`, fallbackStyle);
    } else {
      this.appendResumePlainTextSegments(segments, plainText, fallbackStyle);
    }
    return segments;
  }
  private markdownBlocksCache: Map<string, ResumeMdBlockCacheEntry> = new Map();
  private resumeMarkdownBlocksForItem(item: ResumeContentItem): ResumeMdBlock[] {
    const cached = this.markdownBlocksCache.get(item.id);
    if (cached !== undefined && cached.text === item.text) {
      return cached.blocks;
    }
    const blocks = resolveResumeMarkdownBlocks(item.text);
    this.markdownBlocksCache.set(item.id, { text: item.text, blocks });
    return blocks;
  }

  private resumeMarkdownSingleLinkSegment(block: ResumeMdBlock): ResumeMdInlineSegment | null {
    if (block.inline.length !== 1) {
      return null;
    }
    const segment = block.inline[0];
    if (segment.kind !== RESUME_MD_BLOCK_LINK || segment.href.length === 0) {
      return null;
    }
    return segment;
  }

  @Builder
  private ResumeMarkdownBody(item: ResumeContentItem) {
    Column({ space: 6 }) {
      ForEach(this.resumeMarkdownBlocksForItem(item), (block: ResumeMdBlock) => {
        if (block.kind === RESUME_MD_BLOCK_HEADING) {
          this.ResumeMarkdownHeading(block, item)
        } else if (block.kind === RESUME_MD_BLOCK_CODE) {
          this.ResumeMarkdownCodeBlock(block, item)
        } else if (block.kind === RESUME_MD_BLOCK_LIST) {
          this.ResumeMarkdownList(block, item)
        } else if (block.kind === RESUME_MD_BLOCK_QUOTE) {
          this.ResumeMarkdownQuote(block, item)
        } else if (block.kind === RESUME_MD_BLOCK_DIVIDER) {
          this.ResumeMarkdownDivider()
        } else {
          this.ResumeMarkdownParagraph(block, item)
        }
      }, (block: ResumeMdBlock) => block.id)
    }
      .width('100%')
  }

const DARK_MODE_PREFERENCES_NAME: string = 'codex_router_dark_mode';
const DARK_MODE_PREFERENCES_KEY: string = 'dark_mode';
export default class EntryAbility extends UIAbility {
  private mainWindow: window.Window | null = null;
  private keyboardHeightChangeCallback: ((height: number) => void) | null = null;
  private avoidAreaChangeCallback: ((options: window.AvoidAreaOptions) => void) | null = null;

  onCreate(want: Want, launchParam: AbilityConstant.LaunchParam): void {
    this.applySavedColorMode();
    hilog.info(DOMAIN, 'testTag', '%{public}s', 'Ability onCreate');
  }
  private applySavedColorMode(): void {
    let saved: preferences.ValueType = '';
    try {
      const store = preferences.getPreferencesSync(this.context, { name: DARK_MODE_PREFERENCES_NAME });
      saved = store.getSync(DARK_MODE_PREFERENCES_KEY, '');
    } catch (err) {
      hilog.error(DOMAIN, 'testTag', 'Failed to read color mode preference. Cause: %{public}s', JSON.stringify(err));
    }
    try {
      if (saved === 'dark') {
        this.context.getApplicationContext().setColorMode(ConfigurationConstant.ColorMode.COLOR_MODE_DARK);
      } else if (saved === 'light') {
        this.context.getApplicationContext().setColorMode(ConfigurationConstant.ColorMode.COLOR_MODE_LIGHT);
      } else {

  }

  }

  private async applyImmersiveWindow(mainWindow: window.Window): Promise<void> {
    try {
      await mainWindow.setWindowLayoutFullScreen(true);
      await mainWindow.setWindowSystemBarEnable([]);
    } catch (err) {
      hilog.error(DOMAIN, 'testTag', 'Failed to hide system bars. Cause: %{public}s', JSON.stringify(err));
    }
  }
  private watchKeyboardHeight(mainWindow: window.Window): void {
    AppStorage.setOrCreate<number>('aiRouterKeyboardHeightPx', 0);
    this.keyboardHeightChangeCallback = (height: number): void => {
      this.updateKeyboardHeight(Math.max(0, height), 'height');
    };
    this.avoidAreaChangeCallback = (options: window.AvoidAreaOptions): void => {
      if (options.type !== window.AvoidAreaType.TYPE_KEYBOARD) {
        return;
      }
      this.updateKeyboardHeight(this.resolveKeyboardAvoidAreaHeight(options.area), 'avoidArea');
    };
    this.keyboardDidShowCallback = (info: window.KeyboardInfo): void => {
      this.updateKeyboardHeight(Math.max(0, info.endRect.height), 'didShow');
    };
    this.keyboardDidHideCallback = (_info: window.KeyboardInfo): void => {
      this.updateKeyboardHeight(0, 'didHide');
    };
    mainWindow.on('keyboardHeightChange', this.keyboardHeightChangeCallback);
    mainWindow.on('avoidAreaChange', this.avoidAreaChangeCallback);
    try {
      mainWindow.on('keyboardDidShow', this.keyboardDidShowCallback);
      mainWindow.on('keyboardDidHide', this.keyboardDidHideCallback);
    } catch (err) {
      hilog.error(DOMAIN, 'testTag', 'Failed to watch keyboard animation. Cause: %{public}s', JSON.stringify(err));
    }
    try {
      this.updateKeyboardHeight(this.resolveKeyboardAvoidAreaHeight(
        mainWindow.getWindowAvoidArea(window.AvoidAreaType.TYPE_KEYBOARD)), 'initialAvoidArea');
    } catch (err) {
      hilog.error(DOMAIN, 'testTag', 'Failed to read keyboard avoid area. Cause: %{public}s', JSON.stringify(err));
    }
  }
  private unwatchKeyboardHeight(): void {
    if (this.mainWindow !== null && this.keyboardHeightChangeCallback !== null) {
      try {
        this.mainWindow.off('keyboardHeightChange', this.keyboardHeightChangeCallback);
      } catch (err) {
        hilog.error(DOMAIN, 'testTag', 'Failed to unwatch keyboard height. Cause: %{public}s', JSON.stringify(err));
      }
    }
    if (this.mainWindow !== null && this.avoidAreaChangeCallback !== null) {
      try {
        this.mainWindow.off('avoidAreaChange', this.avoidAreaChangeCallback);
      } catch (err) {
        hilog.error(DOMAIN, 'testTag', 'Failed to unwatch keyboard avoid area. Cause: %{public}s',
          JSON.stringify(err));
      }
    }
    if (this.mainWindow !== null && this.keyboardDidShowCallback !== null) {
      try {
        this.mainWindow.off('keyboardDidShow', this.keyboardDidShowCallback);
      } catch (err) {
        hilog.error(DOMAIN, 'testTag', 'Failed to unwatch keyboard show. Cause: %{public}s', JSON.stringify(err));
      }
    }
    if (this.mainWindow !== null && this.keyboardDidHideCallback !== null) {
      try {
        this.mainWindow.off('keyboardDidHide', this.keyboardDidHideCallback);
      } catch (err) {
        hilog.error(DOMAIN, 'testTag', 'Failed to unwatch keyboard hide. Cause: %{public}s', JSON.stringify(err));
      }
    }
    this.keyboardHeightChangeCallback = null;
    this.avoidAreaChangeCallback = null;
    this.keyboardDidShowCallback = null;
    this.keyboardDidHideCallback = null;
    AppStorage.setOrCreate<number>('aiRouterKeyboardHeightPx', 0);
    AppStorage.set<number>('aiRouterKeyboardHeightPx', 0);
  }
  private resolveKeyboardAvoidAreaHeight(area: window.AvoidArea): number {
    if (!area.visible) {
      return 0;
    }
    return Math.max(0, area.bottomRect.height);
  }
  private updateKeyboardHeight(height: number, source: string): void {
    const nextHeight = Math.max(0, height);
    AppStorage.setOrCreate<number>('aiRouterKeyboardHeightPx', nextHeight);
    AppStorage.set<number>('aiRouterKeyboardHeightPx', nextHeight);
    hilog.info(DOMAIN, 'testTag', 'Keyboard height changed by %{public}s: %{public}d', source, nextHeight);
  }
}

export function normalizeServerAddress(input: string): NormalizedServerAddress {
  let trimmedInput = input.trim();
  let scheme = DEFAULT_SCHEME;
  let address = trimmedInput;
  const schemeSeparatorIndex = trimmedInput.indexOf('://');
  if (schemeSeparatorIndex >= 0) {
    scheme = trimmedInput.substring(0, schemeSeparatorIndex);
    address = trimmedInput.substring(schemeSeparatorIndex + 3);
  }
  let host = address;
  let port = DEFAULT_PORT;
  const portSeparatorIndex = address.lastIndexOf(':');
  if (portSeparatorIndex > -1) {
    const portText = address.substring(portSeparatorIndex + 1);
    const parsedPort = Number(portText);
    if (parsedPort === parsedPort && parsedPort > 0 && parsedPort <= 65535) {
      host = address.substring(0, portSeparatorIndex);
      port = parsedPort;
    }
  }
  return {
    scheme: scheme,
    host: host,
    port: port,
    baseUrl: scheme + '://' + host + ':' + port.toString()
  };
}

export function resolveCarouselIndex(currentIndex: number, itemCount: number): number {
  if (itemCount <= 0) {
    return 0;

    return itemCount - 1;
  }

export function resolveCarouselSwipeIndex(currentIndex: number, itemCount: number, offsetX: number): number {
  if (itemCount <= 1) {
    return 0;
  }
  const threshold = 24;
  if (offsetX < -threshold) {
    return currentIndex >= itemCount - 1 ? 0 : currentIndex + 1;
  }
  if (offsetX > threshold) {
    return currentIndex <= 0 ? itemCount - 1 : currentIndex - 1;
  }
  return resolveCarouselIndex(currentIndex, itemCount);
}

export function resolveConnectionSwiperReleaseIndex(currentIndex: number, itemCount: number, position: number): number {

  if (position < 0) {
    return current >= itemCount - 1 ? 0 : current + 1;

  if (position > 0) {
    return current <= 0 ? itemCount - 1 : current - 1;
  }
  return current;
}

export function resolveConnectionCardSwitchOffset(position: number, gapReduction: number): number {
  if (gapReduction <= 0 || position === 0 || Math.abs(position) >= 1) {
    return 0;

  const normalizedPosition = Math.abs(position);
  const switchProgress = normalizedPosition <= 0.5 ? normalizedPosition * 2 : (1 - normalizedPosition) * 2;
  const offset = Math.round((gapReduction / 2) * switchProgress * 1000) / 1000;
  return position < 0 ? offset : -offset;
}

export function resolveAddConnectionCardOffset(
  pageHeight: number,
  cardHeight: number,
  keyboardGap: number,
  keyboardHeight: number = 0
): number {
  if (pageHeight <= 0 || cardHeight <= 0) {
    return 0;
  }
  const visibleHeight = keyboardHeight > 0 && pageHeight > keyboardHeight ? pageHeight - keyboardHeight : pageHeight;
  const centeredTop = (pageHeight - cardHeight) / 2;
  const safeBottom = visibleHeight - keyboardGap;
  const centeredBottom = centeredTop + cardHeight;
  if (centeredBottom <= safeBottom) {
    return 0;
  }
  return safeBottom - centeredBottom;
}

  return `${twoDigit(date.getMonth() + 1)}-${twoDigit(date.getDate())} ` +

export function resolveResumeDiffLineContext(line: string, inFileDiffContext: boolean): boolean {
  const normalizedLine = line.trim();
  if (normalizedLine.length === 0) {
    return false;
  }
  const lowerLine = normalizedLine.toLowerCase();
  if (lowerLine.startsWith('diff --git ') || lowerLine === '```diff' || lowerLine.startsWith('file ') ||
    lowerLine.startsWith('file:') || lowerLine.startsWith('文件 ') || lowerLine.startsWith('文件:') ||
    normalizedLine.startsWith('--- ') || normalizedLine.startsWith('+++ ') || normalizedLine.startsWith('@@')) {
    return true;
  }
  if (!inFileDiffContext) {
    return false;
  }
  const marker = line.charAt(0);
  return marker === '-' || marker === '+' || marker === ' ' || normalizedLine.startsWith('\\ No newline');
}

const CONNECTION_PREFERENCES_NAME: string = 'codex_router_connections';
const CONNECTION_PREFERENCES_KEY: string = 'connections';
interface StoredConnectionsPayload {
  version?: number;
  connections?: SavedConnection[];
}
export interface ConnectionRepository {
  list(): SavedConnection[];
  add(connection: SavedConnection): boolean;
  update(connection: SavedConnection): boolean;
  delete(id: string): boolean;
}
export class InMemoryConnectionRepository implements ConnectionRepository {
  private connections: SavedConnection[] = [];
  list(): SavedConnection[] {
    return sortConnectionsByLastUsed(this.connections);
  }
  add(connection: SavedConnection): boolean {
    if (!canAddConnection(this.connections)) {
      return false;
    }
    this.connections = sortConnectionsByLastUsed(this.connections.concat([connection]));
    return true;
  }
  update(connection: SavedConnection): boolean {
    let found = false;
    this.connections = this.connections.map((item: SavedConnection) => {
      if (item.id !== connection.id) {
        return item;
      }
      found = true;
      return connection;
    });
    if (found) {
      this.connections = sortConnectionsByLastUsed(this.connections);
    }
    return found;
  }
  delete(id: string): boolean {
    const before = this.connections.length;
    this.connections = this.connections.filter((item: SavedConnection) => item.id !== id);
    return this.connections.length !== before;
  }
}
export interface PersistentConnectionRepository {
  list(): Promise<SavedConnection[]>;
  save(connections: SavedConnection[]): Promise<void>;
}
export class PreferencesConnectionRepository implements PersistentConnectionRepository {
  private context: common.Context;
  private store: preferences.Preferences | null = null;
  constructor(context: common.Context) {
    this.context = context;
  }
  async list(): Promise<SavedConnection[]> {
    try {
      const store = await this.getStore();
      const saved = await store.get(CONNECTION_PREFERENCES_KEY, '');
      if (typeof saved !== 'string') {
        return [];
      }
      let plainValue: string;
      let needsMigration: boolean = false;
      if (isEncryptedConnectionStore(saved)) {
        const decrypted = await decryptConnectionStore(saved);
        if (decrypted === null) {
          return [];
        }
        plainValue = decrypted;
      } else {
        plainValue = saved;
        needsMigration = true;
      }
      const connections = deserializeSavedConnections(plainValue);
      const normalizedSaved = serializeSavedConnections(connections);
      if (normalizedSaved !== plainValue || needsMigration) {
        try {
          await this.save(connections);
        } catch (error) {
        }
      }
      return connections;
    } catch (error) {
      return [];
    }
  }
  async save(connections: SavedConnection[]): Promise<void> {
    try {
      const store = await this.getStore();
      const encrypted = await encryptConnectionStore(serializeSavedConnections(connections));
      await store.put(CONNECTION_PREFERENCES_KEY, encrypted);
      await store.flush();
    } catch (error) {
      throw new Error('Failed to save connections');
    }
  }
  private async getStore(): Promise<preferences.Preferences> {
    if (this.store !== null) {
      return this.store;
    }
    try {
      this.store = await preferences.getPreferences(this.context, CONNECTION_PREFERENCES_NAME);
      return this.store;
    } catch (error) {
      throw new Error('Failed to open connection preferences');
    }
  }
}
export function serializeSavedConnections(connections: SavedConnection[]): string {
  const payload: StoredConnectionsPayload = {
    version: 1,
    connections: uniqueConnectionHosts(sortConnectionsByLastUsed(connections).map((connection: SavedConnection) => {
      return normalizeConnectionId(connection);
    }))
      .slice(0, MAX_CONNECTIONS)
  };
  return JSON.stringify(payload);
}
export function deserializeSavedConnections(value: string): SavedConnection[] {
  if (value.trim().length === 0) {
    return [];
  }
  try {
    const payload: StoredConnectionsPayload = JSON.parse(value) as StoredConnectionsPayload;
    let source: SavedConnection[] = [];
    if (Array.isArray(payload.connections)) {
      source = payload.connections as SavedConnection[];
    }
    const normalized: SavedConnection[] = [];
    source.forEach((item: SavedConnection) => {
      const connection = normalizeStoredConnection(item);
      if (connection !== null) {
        normalized.push(connection);
      }
    });
    return uniqueConnectionHosts(sortConnectionsByLastUsed(normalized).map((connection: SavedConnection) => {
      return normalizeConnectionId(connection);
    }))
      .slice(0, MAX_CONNECTIONS);
  } catch (error) {
    return [];
  }
}
function normalizeStoredConnection(item: SavedConnection): SavedConnection | null {
  if (item === null || item === undefined) {
    return null;
  }
  if (typeof item.host !== 'string' || item.host.trim().length === 0) {
    return null;
  }
  const scheme = typeof item.scheme === 'string' && item.scheme.trim().length > 0 ? item.scheme.trim() : 'https';
  const port = typeof item.port === 'number' && item.port > 0 && item.port <= 65535 ? item.port : 8443;
  const lastUsedAt = normalizeLastUsedAt(item.lastUsedAt);
  return normalizeConnectionId({
    id: typeof item.id === 'string' ? item.id : '',
    name: typeof item.name === 'string' && item.name.trim().length > 0 ? item.name : item.host,
    agent: 'OpenCode',
    host: item.host,
    port,
    scheme,
    username: typeof item.username === 'string' ? item.username : '',
    lastUsedAt,
    status: normalizeStoredStatus(item.status),
    accessTokenRef: typeof item.accessTokenRef === 'string' ? item.accessTokenRef : '',
    refreshTokenRef: typeof item.refreshTokenRef === 'string' ? item.refreshTokenRef : '',
    privateKeyRef: typeof item.privateKeyRef === 'string' ? item.privateKeyRef : ''
  });
}
function normalizeStoredStatus(status: ConnectionStatus): ConnectionStatus {
  if (status === 'needs_login') {
    return 'needs_login';
  }
  return 'offline';
}
function normalizeLastUsedAt(value: string): string {
  if (typeof value !== 'string') {
    return '1970-01-01T00:00:00.000Z';
  }
  const parsedAt = Date.parse(value);
  if (parsedAt !== parsedAt) {
    return '1970-01-01T00:00:00.000Z';
  }
  return value;
}
function normalizeConnectionId(connection: SavedConnection): SavedConnection {
  return {
    id: generateSavedConnectionId(connection.host),
    name: connection.name,
    agent: connection.agent,
    host: connection.host,
    port: connection.port,
    scheme: connection.scheme,
    username: connection.username,
    lastUsedAt: connection.lastUsedAt,
    status: connection.status,
    accessTokenRef: connection.accessTokenRef,
    refreshTokenRef: connection.refreshTokenRef,
    privateKeyRef: connection.privateKeyRef
  };
}
function uniqueConnectionHosts(connections: SavedConnection[]): SavedConnection[] {
  const hosts: string[] = [];
  const result: SavedConnection[] = [];
  connections.forEach((connection: SavedConnection) => {
    const host = `${'OpenCode'}:${connection.host.trim().toLowerCase()}`;
    if (hosts.indexOf(host) >= 0) {
      return;
    }
    hosts.push(host);
    result.push(connection);
  });
  return result;
}

function generateUUID(): string {
  const chars = '0123456789abcdef';
  let uuid = '';
  for (let i = 0; i < 36; i++) {
    if (i === 8 || i === 13 || i === 18 || i === 23) {
      uuid += '-';
    } else {
      const randomIndex = Math.floor(Math.random() * 16);
      uuid += chars.charAt(randomIndex);
    }
  }
  return uuid;
}

  if (cachedDeviceId.length > 0) {
    return cachedDeviceId;
  }
  try {
    const store = await preferences.getPreferences(context, DEVICE_ID_PREFS_NAME);
    const saved = await store.get(DEVICE_ID_PREFS_KEY, '');
    if (typeof saved === 'string' && saved.trim().length > 0) {
      cachedDeviceId = saved.trim();
      return cachedDeviceId;
    }
  } catch (error) {
  }
  const newId = generateUUID();
  cachedDeviceId = newId;
  try {
    const store = await preferences.getPreferences(context, DEVICE_ID_PREFS_NAME);
    await store.put(DEVICE_ID_PREFS_KEY, newId);
    await store.flush();
  } catch (error) {
  }
  return newId;
}

function stringToUint8Array(str: string): Uint8Array {
  const result: number[] = [];
  for (let i = 0; i < str.length; i++) {
    result.push(str.charCodeAt(i) & 0xff);
  }
  return new Uint8Array(result);
}
function uint8ArrayToBase64(data: Uint8Array): string {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';
  let result = '';
  for (let i = 0; i < data.length; i += 3) {
    const b0 = data[i];
    const b1 = i + 1 < data.length ? data[i + 1] : 0;
    const b2 = i + 2 < data.length ? data[i + 2] : 0;
    result += chars.charAt(b0 >> 2);
    result += chars.charAt(((b0 & 3) << 4) | (b1 >> 4));
    result += i + 1 < data.length ? chars.charAt(((b1 & 15) << 2) | (b2 >> 6)) : '=';
    result += i + 2 < data.length ? chars.charAt(b2 & 63) : '=';
  }
  return result;
}
function wrapDerInPem(derData: Uint8Array): string {
  const b64 = uint8ArrayToBase64(derData);
  const lines: string[] = [];
  for (let i = 0; i < b64.length; i += 64) {
    lines.push(b64.substring(i, i + 64));
  }
  return `-----BEGIN PUBLIC KEY-----\n${lines.join('\n')}\n-----END PUBLIC KEY-----`;
}
function buildEd25519SpkiDer(rawPublicKey: Uint8Array): Uint8Array {
  let keyBytes = rawPublicKey;
  if (keyBytes.length === 33 && keyBytes[0] === 0x00) {
    keyBytes = keyBytes.slice(1);
  }
  const algorithmId = new Uint8Array([0x30, 0x05, 0x06, 0x03, 0x2b, 0x65, 0x70]);
  const bitStringValue = new Uint8Array(1 + keyBytes.length);
  bitStringValue[0] = 0x00;
  bitStringValue.set(keyBytes, 1);
  const bitString = new Uint8Array(2 + bitStringValue.length);
  bitString[0] = 0x03;
  bitString[1] = bitStringValue.length;
  bitString.set(bitStringValue, 2);
  const contentLen = algorithmId.length + bitString.length;
  const sequence = new Uint8Array(2 + contentLen);
  sequence[0] = 0x30;
  sequence[1] = contentLen;
  sequence.set(algorithmId, 2);
  sequence.set(bitString, 2 + algorithmId.length);
  return sequence;
}

const CONNECTION_KEY_ALIAS: string = 'codex_router_connection_store_key';
const ENCRYPTED_STORE_PREFIX: string = 'ENC1:';
const GCM_TAG_LENGTH: number = 16;
const GCM_NONCE_LENGTH: number = 12;
function connectionKeyParams(): huks.HuksParam[] {
  return [
    { tag: huks.HuksTag.HUKS_TAG_ALGORITHM, value: huks.HuksKeyAlg.HUKS_ALG_AES },
    { tag: huks.HuksTag.HUKS_TAG_KEY_SIZE, value: huks.HuksKeySize.HUKS_AES_KEY_SIZE_256 },
    {
      tag: huks.HuksTag.HUKS_TAG_PURPOSE,
      value: huks.HuksKeyPurpose.HUKS_KEY_PURPOSE_ENCRYPT | huks.HuksKeyPurpose.HUKS_KEY_PURPOSE_DECRYPT
    },
    { tag: huks.HuksTag.HUKS_TAG_PADDING, value: huks.HuksKeyPadding.HUKS_PADDING_NONE },
    { tag: huks.HuksTag.HUKS_TAG_BLOCK_MODE, value: huks.HuksCipherMode.HUKS_MODE_GCM }
  ];
}
function cipherParams(purpose: huks.HuksKeyPurpose, nonce: Uint8Array | null, tag: Uint8Array | null): huks.HuksParam[] {
  const params: huks.HuksParam[] = [
    { tag: huks.HuksTag.HUKS_TAG_ALGORITHM, value: huks.HuksKeyAlg.HUKS_ALG_AES },
    { tag: huks.HuksTag.HUKS_TAG_KEY_SIZE, value: huks.HuksKeySize.HUKS_AES_KEY_SIZE_256 },
    { tag: huks.HuksTag.HUKS_TAG_PURPOSE, value: purpose },
    { tag: huks.HuksTag.HUKS_TAG_PADDING, value: huks.HuksKeyPadding.HUKS_PADDING_NONE },
    { tag: huks.HuksTag.HUKS_TAG_BLOCK_MODE, value: huks.HuksCipherMode.HUKS_MODE_GCM }
  ];
  if (nonce !== null) {
    params.push({ tag: huks.HuksTag.HUKS_TAG_NONCE, value: nonce });
  }
  if (tag !== null) {
    params.push({ tag: huks.HuksTag.HUKS_TAG_AE_TAG, value: tag });
  }
  return params;
}
async function ensureConnectionKey(): Promise<void> {
  try {
    await huks.generateKeyItem(CONNECTION_KEY_ALIAS, { properties: connectionKeyParams() });
  } catch (error) {
  }
}
function utf8Encode(value: string): Uint8Array {
  const textEncoder = new util.TextEncoder();
  return textEncoder.encodeInto(value);
}
function utf8Decode(value: Uint8Array): string {
  const textDecoder = new util.TextDecoder('utf-8');
  return textDecoder.decodeToString(value);
}
function copyBufferToBytes(value: buffer.Buffer): Uint8Array {
  const bytes = new Uint8Array(value.length);
  for (let i = 0; i < value.length; i++) {
    bytes[i] = value[i];
  }
  return bytes;
}
export function isEncryptedConnectionStore(storedValue: string): boolean {
  return storedValue.startsWith(ENCRYPTED_STORE_PREFIX);
}
export async function encryptConnectionStore(plainText: string): Promise<string> {
  try {
    const options: huks.HuksOptions = {
      properties: cipherParams(huks.HuksKeyPurpose.HUKS_KEY_PURPOSE_ENCRYPT, null, null),
      inData: utf8Encode(plainText)
    };
    await ensureConnectionKey();
    const handle = await huks.initSession(CONNECTION_KEY_ALIAS, options);
    const finish = await huks.finishSession(handle.handle, options);
    const outData = finish.outData;
    if (outData === undefined || outData.length === 0) {
      throw new Error('Failed to encrypt connection store');
    }
    return ENCRYPTED_STORE_PREFIX + buffer.from(outData).toString('base64');
  } catch (error) {
    throw new Error('Failed to encrypt connection store');
  }
}
export async function decryptConnectionStore(storedValue: string): Promise<string | null> {
  if (!isEncryptedConnectionStore(storedValue)) {
    return null;
  }
  let blob: Uint8Array;
  try {
    const decoded = buffer.from(storedValue.substring(ENCRYPTED_STORE_PREFIX.length), 'base64');
    blob = copyBufferToBytes(decoded);
  } catch (error) {
    return null;
  }
  const minimumLength: number = GCM_NONCE_LENGTH + GCM_TAG_LENGTH;
  if (blob.length < minimumLength) {
    return null;
  }
  const nonce = blob.subarray(blob.length - GCM_NONCE_LENGTH);
  const tag = blob.subarray(blob.length - GCM_NONCE_LENGTH - GCM_TAG_LENGTH, blob.length - GCM_NONCE_LENGTH);
  const cipher = blob.subarray(0, blob.length - GCM_NONCE_LENGTH - GCM_TAG_LENGTH);
  const options: huks.HuksOptions = {
    properties: cipherParams(huks.HuksKeyPurpose.HUKS_KEY_PURPOSE_DECRYPT, nonce, tag),
    inData: cipher
  };
  try {
    const handle = await huks.initSession(CONNECTION_KEY_ALIAS, options);
    const finish = await huks.finishSession(handle.handle, options);
    const outData = finish.outData;
    if (outData === undefined || outData.length === 0) {
      return null;
    }
    return utf8Decode(outData);
  } catch (error) {
    return null;
  }
}
