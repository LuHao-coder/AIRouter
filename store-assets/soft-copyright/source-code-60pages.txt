  if (!(error instanceof GatewayHttpError)) {
    return false;
  }
  return error.statusCode === 401 || error.statusCode === 403;
}

class SystemGatewayTransport implements GatewayTransport {
  private session: rcp.Session;

  constructor(caContent: string = '') {
    this.caContent = caContent;
    const timeoutCfg: rcp.Timeout = {
      connectMs: 30000,
      transferMs: 30000
    };
    if (caContent.length > 0) {
      console.info(`[SystemGatewayTransport] Creating session with CA cert, length=${caContent.length}`);
      const caCert: rcp.CertificateAuthority = { content: this.caContent };
      const sessionCfg: rcp.SessionConfiguration = {
        requestConfiguration: {
          transfer: {
            timeout: timeoutCfg
          },
          security: { remoteValidation: caCert }
        }
      };
      this.session = rcp.createSession(sessionCfg);
    } else {
      console.info('[SystemGatewayTransport] Creating session WITHOUT CA cert (default trust)');
      const sessionCfg: rcp.SessionConfiguration = {
        requestConfiguration: {
          transfer: {
            timeout: timeoutCfg
          }
        }
      };
      this.session = rcp.createSession(sessionCfg);
    }
  }

function generateUUID(): string {
  const chars = '0123456789abcdef';
  let uuid = '';
  for (let i = 0; i < 36; i++) {
    if (i === 8 || i === 13 || i === 18 || i === 23) {
      uuid += '-';
    } else {
      const randomIndex = Math.floor(Math.random() * 16);
      uuid += chars.charAt(randomIndex);
    }
  }
  return uuid;
}

function buildEd25519SpkiDer(rawPublicKey: Uint8Array): Uint8Array {
  let keyBytes = rawPublicKey;
  if (keyBytes.length === 33 && keyBytes[0] === 0x00) {
    keyBytes = keyBytes.slice(1);
  }
  const algorithmId = new Uint8Array([0x30, 0x05, 0x06, 0x03, 0x2b, 0x65, 0x70]);
  const bitStringValue = new Uint8Array(1 + keyBytes.length);
  bitStringValue[0] = 0x00;
  bitStringValue.set(keyBytes, 1);
  const bitString = new Uint8Array(2 + bitStringValue.length);
  bitString[0] = 0x03;
  bitString[1] = bitStringValue.length;
  bitString.set(bitStringValue, 2);
  const contentLen = algorithmId.length + bitString.length;
  const sequence = new Uint8Array(2 + contentLen);
  sequence[0] = 0x30;
  sequence[1] = contentLen;
  sequence.set(algorithmId, 2);
  sequence.set(bitString, 2 + algorithmId.length);
  return sequence;
}

export async function decryptConnectionStore(storedValue: string): Promise<string | null> {
  if (!isEncryptedConnectionStore(storedValue)) {
    return null;
  }
  let blob: Uint8Array;
  try {
    const decoded = buffer.from(storedValue.substring(ENCRYPTED_STORE_PREFIX.length), 'base64');
    blob = copyBufferToBytes(decoded);
  } catch (error) {
    return null;
  }
  const minimumLength: number = GCM_NONCE_LENGTH + GCM_TAG_LENGTH;
  if (blob.length < minimumLength) {
    return null;
  }
  const nonce = blob.subarray(blob.length - GCM_NONCE_LENGTH);
  const tag = blob.subarray(blob.length - GCM_NONCE_LENGTH - GCM_TAG_LENGTH, blob.length - GCM_NONCE_LENGTH);
  const cipher = blob.subarray(0, blob.length - GCM_NONCE_LENGTH - GCM_TAG_LENGTH);
  const options: huks.HuksOptions = {
    properties: cipherParams(huks.HuksKeyPurpose.HUKS_KEY_PURPOSE_DECRYPT, nonce, tag),
    inData: cipher
  };
  try {
    const handle = await huks.initSession(CONNECTION_KEY_ALIAS, options);
    const finish = await huks.finishSession(handle.handle, options);
    const outData = finish.outData;
    if (outData === undefined || outData.length === 0) {
      return null;
    }
    return utf8Decode(outData);
  } catch (error) {
    return null;
  }
